IN
CALL fib
OUT
HLT

fib:
    POP AX

    PUSH AX
    PUSH 1
    JE if_n_one

    PUSH AX
    PUSH 2
    JE if_n_is_two

    PUSH AX
    PUSH AX
    PUSH 1
    SUB
    CALL fib

    POP BX
    POP AX
    PUSH BX
    PUSH AX
    PUSH 2
    SUB
    CALL fib

    ADD
    RET

if_n_one:
    PUSH 0
    RET

if_n_is_two:
    PUSH 1
    RET
